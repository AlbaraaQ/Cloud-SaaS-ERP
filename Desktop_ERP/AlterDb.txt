CREATE PROCEDURE [dbo].[GetSalesByPaymentMethods]
	@branch  [int] ,
	@inv_type  [int] ,
	@StartDate  dateTime ,
	@EndDate  dateTime 

WITH EXECUTE AS CALLER
AS
BEGIN
select '''' as PayMethod, PayTypeID, sum(Sales) as TotalSales,sum(ReturnSales) as ReturnSales , sum(Sales-ReturnSales) as SaleNet  from
(select pay_type as PayTypeID , count(*) as InvCount,CASE

WHEN  pay_type = 2 THEN   sum(visa) 
WHEN  pay_type = 4 THEN   sum(visa + cash)
WHEN  pay_type = -1 THEN  sum(tot_net) 
End as Sales ,0 as ReturnSales
from inv where (@branch is null or branch =@branch ) and (inv_type=2 or inv_type=3) and proc_type=1 and (@StartDate is null or date&gt;=@StartDate)and (@EndDate is null or date&lt;=@EndDate)group by pay_type

WHEN  pay_type = 1  THEN  sum(tot_net) 
WHEN  pay_type = 2 THEN   sum(visa) 
WHEN  pay_type = 4 THEN   sum(visa + cash)
WHEN  pay_type = -1 THEN  sum(tot_net) 
End as ReturnSales
from inv where (@branch is null or branch =@branch ) and (inv_type=2 or inv_type=3) and proc_type=2 and (@StartDate is null or date&gt;=@StartDate)and (@EndDate is null or date&lt;=@EndDate)group by pay_type   ) as tb group by PayTypeID

CREATE PROCEDURE [dbo].[MonthlySales]
	@branch  [int] ,
	@inv_type  [int] ,
	@pay_type  [int] ,
	@Seller  [int] 
WITH EXECUTE AS CALLER
AS
BEGIN
select InvDate,InvYear,InvMonth , sum(InvCount) as InvCount,sum(InvReturnCount) as InvReturnCount ,sum(InvTotal)as InvTotal , sum(ReturnTotal) as ReturnTotal, sum(InvTotal-ReturnTotal) as SaleNet from
(select  CONCAT(CAST(MONTH(date) AS VARCHAR(2)) , ''-'' , CAST(YEAR(date) AS VARCHAR(4))) as InvDate,0 as InvCount, sum(InvTotal) as InvTotal ,0 as InvReturnCount,0 as ReturnTotal,YEAR(date) as InvYear, month(date) as InvMonth
from inv where (@branch is null or branch =@branch ) and (@inv_type is null or(inv_type=@inv_type or inv_type=@inv_type)) and proc_type=1 and IS_Deleted=0  group by year(date), month(date)
union all
select  CONCAT(CAST(MONTH(date) AS VARCHAR(2)) , ''-'' , CAST(YEAR(date) AS VARCHAR(4))) as InvDate ,0 as InvCount,0 as InvTotal, count(*) as InvReturnCount, sum(InvTotal) as ReturnTotal ,YEAR(date) as InvYear, month(date) as InvMonth
from inv where (@branch is null or branch =@branch ) and (@inv_type is null or(inv_type=@inv_type or inv_type=@inv_type)) and proc_type=2 and IS_Deleted=0 group by year(date), month(date)) as tb group by InvDate,InvYear,InvMonth order by InvYear,InvMonth
END')

IF NOT EXISTS (SELECT * FROM sys.objects WHERE type = 'P' AND OBJECT_ID = OBJECT_ID('dbo.GetAccountStatement'))
BEGIN
   
   
   
   
   
   EXEC('Create PROCEDURE [dbo].[GetAccountStatement]
    @ParentCode NVARCHAR(50),
    @StartDate DATETIME,
    @EndDate DATETIME,
    @Branch NVARCHAR(50),
    @ShowPreviousBalance BIT
AS
BEGIN
    SET NOCOUNT ON;

    -- CTE to get account hierarchy
    WITH AccountHierarchy AS (
        SELECT 
            Code,
            AName,
            Type,
            ParentCode
        FROM 
            Accounts_Index
        WHERE 
            Code = @ParentCode
        
        UNION ALL
        
        SELECT 
            a.Code,
            a.AName,
            a.Type,
            a.ParentCode
        FROM 
            Accounts_Index a
        INNER JOIN 
            AccountHierarchy ah ON a.ParentCode = ah.Code
    ),

    -- CTE to calculate previous balance for child accounts
    PreviousBalance AS (
        SELECT 
            @ParentCode AS Code,
             SUM(es.dept) as PreviousDept,
			 SUM(es.credit) AS PreviousCredit
        FROM 
            entry e
        INNER JOIN 
            entry_sub es ON e.GlobalID = es.EntryGlobalID
        WHERE 
            es.acc_no IN (SELECT Code FROM AccountHierarchy WHERE Type = 2)
            AND e.is_Deleted = 0 -- Exclude deleted entries if needed
            AND e.date &lt; @StartDate
            AND (@Branch IS NULL OR e.branch = @Branch)
    ),

    -- CTE to get the main statement
    MainStatement AS (
        SELECT 
            ROW_NUMBER() OVER (ORDER BY e.date) AS DgvRank,
            es.acc_no AS DgvAccCode, 
            ah.AName AS DgvAccName,
            e.doc_No AS DgvEntryNo, 
            es.dept AS DgvDept, 
            es.credit AS DgvCredit,
            e.date AS DgvDate,
            es.notes AS DgvNote,
            es.EntryGlobalID AS DgvGenralNo,
            b.name AS DgvBranch,
			0 as DgvBalance,
            '''' DgvBalanceStatus,
            et.Name AS DgvEntryType
        FROM 
            entry e
        INNER JOIN 
            entry_sub es ON e.GlobalID = es.EntryGlobalID
        INNER JOIN 
            AccountHierarchy ah ON es.acc_no = ah.Code
		LEFT JOIN 
            Branches b ON e.branch = b.BranchId
			LEFT JOIN
			EntryTypes et on e.type=et.Id 
        WHERE 
            es.acc_no IN (SELECT Code FROM AccountHierarchy)
            AND e.is_Deleted = 0 -- Exclude deleted entries if needed
            AND (@StartDate IS NULL OR e.date &gt;= @StartDate)
            AND (@EndDate IS NULL OR e.date &lt;= @EndDate)
            AND (@Branch IS NULL OR e.branch = @Branch)
    )

    -- Combine the previous balance with the main statement based on @ShowPreviousBalance
    SELECT
        DgvRank,
        DgvAccCode,
        DgvAccName,
        DgvEntryNo,
        DgvDept,
        DgvCredit,
        DgvDate,
        DgvNote,
        DgvGenralNo,
        DgvBranch,
        DgvBalance,
        DgvBalanceStatus,
        DgvEntryType
    FROM (
        -- Previous balance row
        SELECT
            0 AS DgvRank, -- Use -1 to ensure it appears first
            pb.Code AS DgvAccCode,
            ai.AName AS DgvAccName,
            NULL AS DgvEntryNo,
            pb.PreviousDept AS DgvDept,
            pb.PreviousCredit AS DgvCredit,
            @StartDate AS DgvDate,
            ''Previous Balance'' AS DgvNote,
            NULL AS DgvGenralNo,
            NULL AS DgvBranch,
            0 AS DgvBalance,
		  ''Previous Balance'' DgvBalanceStatus,
            ''Previous Balance'' AS DgvEntryType
        FROM
            PreviousBalance pb
        INNER JOIN
            Accounts_Index ai ON pb.Code = ai.Code
        WHERE @ShowPreviousBalance = 1 

        UNION ALL

        -- Main statement rows
        SELECT
            ms.DgvRank,
            ms.DgvAccCode,
            ms.DgvAccName,
            ms.DgvEntryNo,
            ms.DgvDept,
            ms.DgvCredit,
            ms.DgvDate,
            ms.DgvNote,
            ms.DgvGenralNo,
            ms.DgvBranch,
            ms.DgvBalance,
            ms.DgvBalanceStatus,
            ms.DgvEntryType
        FROM
            MainStatement ms
    ) AS CombinedResult
    ORDER BY
        DgvRank, DgvDate;
END;')

IF NOT EXISTS (SELECT 1 FROM sys.tables WHERE name = 'AccountingPeriods')

    PeriodID INT PRIMARY KEY IDENTITY(1,1),
    PeriodName NVARCHAR(100) NOT NULL,
    StartDate DATE NOT NULL,
    EndDate DATE NOT NULL,
    IsClosed BIT DEFAULT 0,
    IsActive BIT DEFAULT 0,
    ClosedBy NVARCHAR(100),
    ClosedDate DATETIME,
    Notes NVARCHAR(500),
    CreatedDate DATETIME DEFAULT GETDATE()
)
End

    LockID INT PRIMARY KEY IDENTITY(1,1),

    ModuleName NVARCHAR(50), -- مثل: Sales, Purchases, Inventory
    IsLocked BIT DEFAULT 0,
    LockedBy NVARCHAR(100),
    LockedDate DATETIME
)
End
